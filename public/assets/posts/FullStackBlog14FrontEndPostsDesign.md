#### 2. Post List Page

**2.1 Create List Page Component**

Create the post list page in `src/app/[locale]/posts/page.tsx`:

```typescript
'use client'
import PostCard from './components/PostCard'
import { usePathname, useRouter, useSearchParams } from 'next/navigation'
import { Pagination } from '@/components/Pagination'
import { useUserStore } from '@/store/userStore'
import { PostCategory, useGetApiPosts } from '@/lib/api/generated'
import { Tabs, TabsList, TabsTrigger } from '@/components/ui/tabs'
import { useTranslations } from 'next-intl'

const pageSize = 10
const postCategories = [PostCategory.life, PostCategory.work, PostCategory.crypto, PostCategory.sports] as const

export default function PagePosts() {
  const t = useTranslations('PostCategory')
  const user = useUserStore((state: any) => state.user)
  const searchParams = useSearchParams()
  const currentPage = Number(searchParams.get('page')) || 1
  const category = searchParams.get('category') || 'all'
  const router = useRouter()
  const pathname = usePathname()

  // Use generated API hook to get post list
  const { data } = useGetApiPosts({
    page: currentPage,
    pageSize: pageSize,
    category: category === 'all' ? undefined : (category as PostCategory),
  })

  // Handle category switching
  const handleCategoryChange = (value: string) => {
    const params = new URLSearchParams(searchParams)
    if (value === 'all') {
      params.delete('category')
    } else {
      params.set('category', value)
    }
    params.set('page', '1') // Reset to first page when switching category
    router.push(`${pathname}?${params.toString()}`)
  }

  // Extract pagination information from response data
  const totalPages = data?.totalCount ? Math.ceil(data?.totalCount / pageSize) : 1
  const posts = data?.items || []

  return (
    <div className="page-wrapper py-6">
      {/* Category filter Tabs */}
      <div className="mb-6 flex justify-start">
        <Tabs value={category} onValueChange={handleCategoryChange}>
          <TabsList className="md:gap-2">
            <TabsTrigger value="all">{t('all')}</TabsTrigger>
            {postCategories.map((cat) => (
              <TabsTrigger key={cat} value={cat}>
                {t(cat)}
              </TabsTrigger>
            ))}
          </TabsList>
        </Tabs>
      </div>

      {/* Post list */}
      <div className="flex flex-col gap-6">
        {posts.map((post: any) => (
          <PostCard key={post.id} post={post} currentUserId={user?.id} />
        ))}
      </div>

      {/* Pagination component */}
      {totalPages > 1 && (
        <div className="mt-8 flex justify-center">
          <Pagination currentPage={currentPage} totalPages={totalPages} />
        </div>
      )}
    </div>
  )
}
```

**Key Features**:

- Use `useGetApiPosts` hook (auto-generated by Orval) to get post list
- Read current page number and category from URL query parameters
- Use Tabs component for category filtering
- Automatically reset to first page when switching category
- Use Pagination component for pagination navigation

**2.2 URL Parameter Management**

- `page`: Current page number (default is 1)
- `category`: Post category (`all`, `life`, `work`, `crypto`, `sports`)
- Use Next.js `useSearchParams` and `useRouter` to manage URL parameters

#### 4. Post Detail Page

**4.1 Create Detail Page Component**

Create the post detail page in `src/app/[locale]/posts/[id]/page.tsx`:

```typescript
import MarkdownView from '@/components/MarkdownView'
import { PostCategory, PostDTO } from '@/lib/api/generated'
import { apiFetchServer } from '@/lib/apiFetch'
import { __IS_PROD__ } from '@/lib/constants'
import { queryClient } from '@/lib/queryClient'
import { format } from 'date-fns'
import { Locale } from 'next-intl'
import { getTranslations, setRequestLocale } from 'next-intl/server'
import PostAdminActions from '../components/PostAdminActions'

type Props = {
  params: Promise<{ locale: Locale; id: string }>
}

export default async function PagePostsDetail({ params }: Props) {
  const { locale, id } = await params
  const t = await getTranslations('PostCategory')
  const tCommon = await getTranslations('Common')
  setRequestLocale(locale)

  // Use queryClient.fetchQuery to get data on server side
  const postDetail: PostDTO = await queryClient.fetchQuery({
    queryKey: ['post', id],
    queryFn: () => apiFetchServer(`/api/posts/${id}`),
  })

  // Try to read content from static files
  let markdownContent = locale === 'zh' ? postDetail.contentZh : postDetail.content

  // Try to generate filename from title and read from AWS S3
  if (postDetail.title) {
    const fileName = generateFileNameFromTitle(postDetail.title)
    const staticContent = await readAwsMarkdown(fileName, locale)
    if (staticContent) {
      markdownContent = staticContent
    }
  }

  return (
    <div className="page-wrapper py-6">
      <div className="">
        <div key={postDetail.id!} className="relative flex flex-col gap-6">
          {/* Post content */}
          <div className="">
            <MarkdownView content={markdownContent || ''} />
          </div>

          {/* Author and publish time */}
          <div className="text-muted-foreground flex items-center gap-6 text-sm">
            <div className="flex items-center gap-2">
              <span>{postDetail.author?.userName}</span>
              <span>{format(postDetail.createdAt!, 'yyyy.MM.dd HH:mm')}</span>
            </div>
            <div>
              {tCommon('category')}: {t(postDetail.category as PostCategory) || ''}
            </div>
          </div>

          {/* Admin actions */}
          <div className="mt-6 flex items-center gap-2">
            <PostAdminActions post={postDetail} />
          </div>
        </div>
      </div>
    </div>
  )
}
```

**Key Features**:

- Server-side rendering: Use `apiFetchServer` to get data on server side
- Use `queryClient.fetchQuery` for data fetching and caching
- Support reading content from database fields or AWS S3 static files
- Display metadata such as author, publish time, category

**4.2 AWS S3 Markdown File Reading**

Implement functionality to read Markdown files from AWS S3:

```typescript
/**
 * Read markdown file content from AWS S3 (via public URL)
 * @param fileName File name (without path and extension)
 * @param locale Locale
 * @returns markdown content, returns null if file doesn't exist
 */
async function readAwsMarkdown(fileName: string, locale: Locale): Promise<string | null> {
  try {
    const baseurl = __IS_PROD__
      ? `${process.env.NEXT_PUBLIC_AWS_S3_ASEETSPREFIX}/public/assets`
      : `${process.env.NEXT_PUBLIC_APP_BASE_URL}/assets`

    // Build file URLs: first try file with locale, then try file without locale
    const filePaths = [`${baseurl}/posts/${fileName}.${locale}.md`, `${baseurl}/posts/${fileName}.md`]

    for (const url of filePaths) {
      try {
        const response = await fetch(url, {
          cache: 'no-cache',
        })

        if (response.ok) {
          const content = await response.text()
          if (content) {
            return content
          }
        } else if (response.status === 404) {
          // File doesn't exist, continue trying next path
          continue
        } else {
          console.warn(`Failed to read file from S3 (${url}): HTTP ${response.status}`)
          continue
        }
      } catch (error: any) {
        console.warn(`Failed to read file from S3 (${url}):`, error.message)
        continue
      }
    }

    return null
  } catch (error: any) {
    console.error('Failed to read markdown file from AWS S3:', error)
    return null
  }
}
```

**File Reading Strategy**:

1. First try to read file with locale (e.g., `FullStackBlog01.zh.md`)
2. If it doesn't exist, try to read file without locale (e.g., `FullStackBlog01.md`)
3. If neither exists, return `null` and use content from database

**4.3 Filename Generation**

Generate filename from post title:

```typescript
/**
 * Generate filename from post title
 * Convert title to filename format (keep alphanumeric, dots, hyphens, and underscores, convert spaces to empty string)
 * Example: "Full Stack Blog 01: Quick Start" -> "FullStackBlog01QuickStart"
 */
function generateFileNameFromTitle(title: string): string {
  return title
    .replace(/[^\w\s.-]/g, '') // Remove special characters
    .replace(/\s+/g, '') // Remove all spaces
    .trim()
}
```

#### 7. Pagination Component

**7.1 Pagination Component Implementation**

Implement the pagination component in `src/components/Pagination.tsx`:

```typescript
'use client'

import {
  Pagination as ShadcnPagination,
  PaginationContent,
  PaginationEllipsis,
  PaginationItem,
  PaginationLink,
  PaginationNext,
  PaginationPrevious,
} from '@/components/ui/pagination'
import { usePathname, useRouter, useSearchParams } from 'next/navigation'
import { MouseEvent } from 'react'

interface PaginationProps {
  currentPage: number
  totalPages: number
}

export function Pagination({ currentPage, totalPages }: PaginationProps) {
  const router = useRouter()
  const pathname = usePathname()
  const searchParams = useSearchParams()

  const createPageURL = (pageNumber: number) => {
    const params = new URLSearchParams(searchParams)
    params.set('page', pageNumber.toString())
    return `${pathname}?${params.toString()}`
  }

  const handlePageChange = (page: number) => {
    router.push(createPageURL(page))
  }

  // Generate page numbers to display (display at most 5 page numbers)
  const getPageNumbers = () => {
    const pages = []
    const maxVisiblePages = 5

    if (totalPages <= maxVisiblePages) {
      // If total pages is less than or equal to max visible pages, display all page numbers
      for (let i = 1; i <= totalPages; i++) {
        pages.push(i)
      }
    } else {
      // Always display first page
      pages.push(1)

      // Calculate start and end of middle page numbers
      let start = Math.max(2, currentPage - 1)
      let end = Math.min(totalPages - 1, currentPage + 1)

      // Adjust start and end page numbers to ensure enough pages are displayed
      if (start > 2) {
        pages.push('...')
      }

      for (let i = start; i <= end; i++) {
        pages.push(i)
      }

      // If last page is not in visible range, add ellipsis
      if (end < totalPages - 1) {
        pages.push('...')
      }

      // Always display last page
      pages.push(totalPages)
    }

    return pages
  }

  return (
    <ShadcnPagination>
      <PaginationContent>
        {/* Previous page button */}
        <PaginationItem>
          <PaginationPrevious
            href="#"
            onClick={(e: MouseEvent) => {
              e.preventDefault()
              if (currentPage > 1) {
                handlePageChange(currentPage - 1)
              }
            }}
            className={currentPage === 1 ? 'pointer-events-none opacity-50' : ''}
          />
        </PaginationItem>

        {/* Page number buttons */}
        {getPageNumbers().map((page, index) => (
          <PaginationItem key={index}>
            {page === '...' ? (
              <PaginationEllipsis />
            ) : (
              <PaginationLink
                href="#"
                onClick={(e: MouseEvent) => {
                  e.preventDefault()
                  handlePageChange(page as number)
                }}
                isActive={currentPage === page}
              >
                {page}
              </PaginationLink>
            )}
          </PaginationItem>
        ))}

        {/* Next page button */}
        <PaginationItem>
          <PaginationNext
            href="#"
            onClick={(e: MouseEvent) => {
              e.preventDefault()
              if (currentPage < totalPages) {
                handlePageChange(currentPage + 1)
              }
            }}
            className={currentPage === totalPages ? 'pointer-events-none opacity-50' : ''}
          />
        </PaginationItem>
      </PaginationContent>
    </ShadcnPagination>
  )
}
```

**Key Features**:

- Smart page number display: Display at most 5 page numbers, use ellipsis to indicate more pages
- Preserve URL query parameters: Preserve category and other parameters when switching pages
- Disabled state: Disable previous page on first page, disable next page on last page
- Use shadcn/ui Pagination component as base

#### 9. Technical Points

**9.1 Server-Side Rendering vs Client-Side Rendering**

- **List Page**: Use client component (`'use client'`), supports interactive operations (category switching, pagination)
- **Detail Page**: Use server component, fetch data on server side, improves SEO and first screen load speed

**9.2 Data Fetching Strategy**

- **List Page**: Use React Query hooks (`useGetApiPosts`), automatically handles caching and refetching
- **Detail Page**: Use `queryClient.fetchQuery`, fetch data on server side and cache

**9.3 Internationalization Implementation**

- Use `next-intl` for internationalization
- Select corresponding fields based on `locale` (`title`/`titleZh`, `content`/`contentZh`)
- Use `Link` component from `@/i18n/navigation` for internationalized routing

**9.4 Markdown Content Sources**

1. **Database Fields**: `content` or `contentZh`
2. **AWS S3 Static Files**: Prioritize reading from S3, if exists then override database content
3. **File Naming Rules**: Generate filename from post title, support files with locale

#### 10. Common Issues

**Q: Why use server component for detail page?**

A: Server components can fetch data on the server side, improving SEO and first screen load speed. Post detail content is relatively static, suitable for server-side rendering.

**Q: How to customize Markdown styles?**

A: Modify `prose` class configuration in `MarkdownView` component, or use Tailwind Typography custom configuration.

**Q: How does pagination component preserve other query parameters?**

A: Use `new URLSearchParams(searchParams)` to create a new parameter object, only modify `page` parameter, other parameters are automatically preserved.
